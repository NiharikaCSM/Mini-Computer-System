% Test 2: sum an array read from data memory, using a loop and branch.
% Data layout (see test2_loop_data.byte):
%   address 0   : array size N
%   address 4.. : N elements, 4 bytes apart
% Expect (in log.txt): running sum printed each iteration, final = 100

x1 = 0      % index
x2 = 0      % running sum
x3 = 0      % address cursor
x4 = [x3]   % x4 = N (size of array)

.loopback
x15 = x1 - x4
BEQ .exit
x3 = x3 + 4
x6 = [x3]
x2 = x2 + x6
x1 = x1 + 1
print x2
BAL .loopback

.exit
[x3] = x2
