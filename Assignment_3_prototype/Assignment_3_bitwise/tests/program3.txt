% Assumes data.byte has: Data[0]=99, Data[4]=0 (scratch), Data[8]=0 (scratch)
x1 = 0
x2 = [x1]        % register-indirect read: x2 = Data[Register[1]] = Data[0] = 99
x3 = [0]         % constant-address read: x3 = Data[0] = 99
x4 = 4
[x4] = x2        % register-indirect write: Data[Register[4]] = x2 -> Data[4] = 99
[8] = x3         % constant-address write: Data[8] = x3 -> Data[8] = 99
read x5, 0       % legacy read: x5 = Data[0] = 99
write x5, 12     % legacy write: Data[12] = x5 = 99
