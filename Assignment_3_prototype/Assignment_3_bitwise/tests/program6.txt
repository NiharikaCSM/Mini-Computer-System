% Multiply two complex numbers (a+bi) * (c+di)
% real = a*c - b*d
% imag = a*d + b*c
% Using a=5, b=2 (i.e. 5+2i) and c=3, d=4 (i.e. 3+4i)
% Expected: real = 15-8 = 7, imag = 20+6 = 26  -> result = 7 + 26i
x1 = 5    % a
x2 = 2    % b
x3 = 3    % c
x4 = 4    % d
 
x5 = x1 * x3     % a*c
x6 = x2 * x4     % b*d
x7 = x5 - x6     % real part = ac - bd
 
x8 = x1 * x4     % a*d
x9 = x2 * x3     % b*c
x10 = x8 + x9    % imag part = ad + bc
 
[0] = x7         % store real part at Data[0]
[4] = x10        % store imaginary part at Data[4]
