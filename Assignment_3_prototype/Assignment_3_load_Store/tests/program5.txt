x1 = 255

% Stores only lower 4 bits into Memory address 10
hb[10] = x1        

% Stores lower byte  into Memory address 11
b[11] = x1         

% Stores lower half-word  into Memory address 12
h[12] = x1         

% Stores full 32-bit word into Memory address 13
[13] = x1          

x2 = hb[10]        
x3 = b[11]        
x4 = h[12]
